! ============================================================
! MLS-1 and MLS-2: access control lists (paste on BOTH switches)
! ============================================================
!
! POLICY
!  - ACLs sit inbound on each VLAN interface (closest to the source).
!  - Users get: DHCP, DNS, ping to own gateway, Internet.
!  - Users never reach: other VLANs, servers (except DNS/DHCP), network devices.
!  - Only the user's own subnet may be a source (anti-spoofing, last permit line).
!  - Admin PC (10.10.50.20) and NMS (10.10.50.11) may reach/monitor user VLANs.
!  - Guest, Customer A and Customer B are Internet-only (no admin/NMS access).
!  - Servers VLAN: replies only, no Internet except Admin PC and DNS forwarding.
!
! BEFORE YOU PASTE
!  1. show standby brief       (VIP of VLAN 100 and 200 must be .1)
!  2. show running-config interface vlan 70
!
! STEP 1: create new ACLs
!
! VLAN 10 Residential
ip access-list extended RES-IN
 permit udp any eq bootpc any eq bootps
 permit udp 192.168.10.0 0.0.0.255 host 10.10.50.12 eq domain
 permit tcp 192.168.10.0 0.0.0.255 host 10.10.50.12 eq domain
 permit icmp 192.168.10.0 0.0.0.255 host 10.10.50.12 echo
 permit icmp 192.168.10.0 0.0.0.255 host 192.168.10.1 echo
 permit icmp 192.168.10.0 0.0.0.255 host 10.10.50.20 echo-reply
 permit icmp 192.168.10.0 0.0.0.255 host 10.10.50.11 echo-reply
 permit tcp 192.168.10.0 0.0.0.255 host 10.10.50.20 established
 permit udp 192.168.10.0 0.0.0.255 eq snmp host 10.10.50.11
 deny ip 192.168.10.0 0.0.0.255 10.0.0.0 0.255.255.255
 deny ip 192.168.10.0 0.0.0.255 172.16.0.0 0.15.255.255
 deny ip 192.168.10.0 0.0.0.255 192.168.0.0 0.0.255.255
 permit ip 192.168.10.0 0.0.0.255 any
 exit

ip access-list extended SERVER-IN
 permit udp 10.10.50.0 0.0.0.255 host 224.0.0.2 eq 1985
 permit ip host 10.10.50.20 any
 permit udp host 10.10.50.11 any eq snmp
 permit icmp host 10.10.50.11 any echo
 permit udp host 10.10.50.12 eq bootps 192.168.0.0 0.0.255.255 eq bootps
 permit udp host 10.10.50.12 eq bootps 192.168.0.0 0.0.255.255 eq bootpc

 permit tcp host 10.10.50.12 eq domain 192.168.0.0 0.0.255.255
 permit udp host 10.10.50.12 any eq domain

! STEP 2: apply inbound
!
interface vlan 10
 ip access-group RES-IN in
interface vlan 20
 ip access-group BIZ-IN in
interface vlan 30

interface vlan 40
 ip access-group GAME-IN in
interface vlan 50

interface vlan 60

interface vlan 100
 ip access-group CUSTA-IN in
interface vlan 200
 ip access-group CUSTB-IN in
interface vlan 70
 ip access-group SERVER-IN in
end
write memory
!
! ROLLBACK (if something breaks, run in config mode)
!  interface vlan 70
!   no ip access-group SERVER-IN in
!  interface vlan 10
